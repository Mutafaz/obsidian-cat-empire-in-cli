    const selRaw = raw.find((c) => c.id === selId) || raw[0];
    const sel = build(selRaw);
    const info = selRaw.locked ? ['Keep hatching eggs to discover this cat!', '???'] : INFO[selRaw.id];
    sel.nameUp = sel.name.toUpperCase();
    sel.nameSize = sel.nameUp.length > 13 ? '34px' : '42px';
    sel.rarityUp = RAR[selRaw.rarity].toUpperCase();
    sel.rosetteSize = sel.rarityUp.length > 6 ? '13px' : '17px';
    sel.rosetteFill = 'url(#rb-' + selRaw.rarity + ')';
    sel.status = selRaw.locked ? 'UNDISCOVERED' : (selRaw.status === 'notowned' ? 'NOT OWNED' : 'DISCOVERED');
    sel.statusColor = selRaw.locked ? '#f87171' : (selRaw.status === 'notowned' ? '#fb923c' : '#4ade80');
    sel.desc = info[0];
    sel.bonus = info[1];
    sel.bonusNote = selRaw.locked ? '(Discover to unlock)' : (selRaw.status === 'notowned' ? '(Collect to activate)' : '(Permanently Active)');
    sel.aria = selRaw.locked ? 'Undiscovered cat silhouette' : selRaw.name;

    const pts = [];
    for (let i = 0; i < 24; i++) {
      const a = -Math.PI / 2 + (i * Math.PI) / 12;
      const r = i % 2 ? 39 : 47;
      pts.push((50 + r * Math.cos(a)).toFixed(1) + ',' + (50 + r * Math.sin(a)).toFixed(1));
    }

    const CAT = [
      { key: 'all', label: 'ALL CATS', top: '#fcd34d', bot: '#e09a12', edge: '#a8650a', ring: '#ffd84a', ringDark: '#b97a0e', inner: '#f5b822', ribbon: '#d98b1a', icon: '#e3a114', icon_t: 'cat', paw: 0 },
      { key: 'common', label: 'COMMON', top: '#c4c8d0', bot: '#7b818d', edge: '#555b66', ring: '#e3e6eb', ringDark: '#7b818d', inner: '#b7bcc6', ribbon: '#8b919c', icon: '#f4f5f7', icon_t: 'star', paw: 1 },
      { key: 'rare', label: 'RARE', top: '#4fc3ff', bot: '#1d7fd6', edge: '#145a9e', ring: '#5cc4ff', ringDark: '#145a9e', inner: '#2b8be0', ribbon: '#1f6fc4', icon: '#c4ebff', icon_t: 'star', paw: 1 },
      { key: 'epic', label: 'EPIC', top: '#c084fc', bot: '#8b3fd9', edge: '#5f2399', ring: '#c98bff', ringDark: '#5f2399', inner: '#9b4de8', ribbon: '#7a2fc7', icon: '#f1ddff', icon_t: 'crown', paw: 1 },
      { key: 'legendary', label: 'LEGENDARY', top: '#ffb347', bot: '#f07a12', edge: '#b05208', ring: '#ffc23d', ringDark: '#b05208', inner: '#f39a1a', ribbon: '#e06a0e', icon: '#fff0b0', icon_t: 'crown', paw: 1 },
      { key: 'mythic', label: 'MYTHIC', top: '#ff7b7b', bot: '#dc2b2b', edge: '#9c1818', ring: '#ff6b6b', ringDark: '#9c1818', inner: '#d92e2e', ribbon: '#b91c1c', icon: '#ffd6d6', icon_t: 'crown', paw: 1 },
      { key: 'secret', label: 'SECRET', top: '#6a5cf0', bot: '#3a2f9e', edge: '#241c6e', ring: '#7c6cf5', ringDark: '#241c6e', inner: '#4433b8', ribbon: '#3b2c9c', icon: '#d6d0ff', icon_t: 'crown', paw: 0 }
    ];
    const categories = CAT.map((k) => {
      const isOn = k.key === active;
      return {
        label: k.label,
        size: k.label.length > 8 ? '27px' : '30px',
        bg: 'linear-gradient(180deg, ' + k.top + ', ' + k.bot + ')',
        shadow: 'inset 0 -7px 0 ' + k.edge + (isOn ? ', 0 0 0 3px #12131a, 0 0 0 7px #ffffff' : ''),
        pressed: isOn ? 'true' : 'false',
        ring: k.ring, ringDark: k.ringDark, inner: k.inner, ribbon: k.ribbon, icon: k.icon,
        starD: on(k.icon_t === 'star'),
        crownD: on(k.icon_t === 'crown'),
        catD: on(k.icon_t === 'cat'),
        pawD: on(k.paw),
        pick: () => this.setState({ cat: k.key })
      };
    });

    return {
      motionClass: motion ? 'motion' : '',
      categories,
      cards,
      empty: cards.length === 0,
      sel,
      rosettePts: pts.join(' '),
      query: s.q || '',
      onSearch: (e) => this.setState({ q: e.target.value }),
      onToggle: () => this.setState({ showLocked: !showLocked }),
      togglePressed: showLocked ? 'true' : 'false',
      trackBg: showLocked ? '#0d3a85' : '#4a4e5a',
      knobLeft: showLocked ? '31px' : '2px'
    };
  }
}
